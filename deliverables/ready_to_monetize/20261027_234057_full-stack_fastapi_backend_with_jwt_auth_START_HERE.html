from collections.abc import AsyncGenerator
from sqlalchemy.ext.asyncio import AsyncSession, async_sessionmaker, create_async_engine
from app.core.config import settings

engine = create_async_engine(settings.DATABASE_URL, echo=False, future=True)

AsyncSessionLocal = async_sessionmaker(
    engine, class_=AsyncSession, expire_on_commit=False
)


async def get_db() -> AsyncGenerator[AsyncSession, None]:
    async with AsyncSessionLocal() as session:
        try:
            yield session
            await session.commit()
        except Exception:
            await session.rollback()
            raise
        finally:
            await session.close()

from datetime import datetime, timedelta
from typing import Any, Union
from jose import jwt
from passlib.context import CryptContext
from app.core.config import settings

pwd_context = CryptContext(schemes=["argon2"], deprecated="auto")


def verify_password(plain_password: str, hashed_password: str) -> bool:
    return pwd_context.verify(plain_password, hashed_password)


def get_password_hash(password: str) -> str:
    return pwd_context.hash(password)


def create_access_token(
    subject: Union[str, Any], expires_delta: timedelta = None
) -> str:
    if expires_delta:
        expire = datetime.utcnow() + expires_delta
    else:
        expire = datetime.utcnow() + timedelta(
            minutes=settings.ACCESS_TOKEN_EXPIRE_MINUTES
        )
    to_encode = {"exp": expire, "sub": str(subject)}
    encoded_jwt = jwt.encode(
        to_encode, settings.SECRET_KEY, algorithm=settings.ALGORITHM
    )
    return encoded_jwt

from fastapi import Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer
from jose import JWTError, jwt
from sqlalchemy.ext.asyncio import AsyncSession
from app.core.config import settings
from app.core.database import get_db
from app.models.user import User
from app.schemas.auth import TokenPayload

reusable_oauth2 = OAuth2PasswordBearer(
    tokenUrl=f"{settings.API_V1_STR}/auth/login"
)


async def get_current_user(
    db: AsyncSession = Depends(get_db), token: str = Depends(reusable_oauth2)
) -> User:
    try:
        payload = jwt.decode(
            token, settings.SECRET_KEY, algorithms=[settings.ALGORITHM]
        )
        token_data = TokenPayload(**payload)
    except (JWTError, ValueError):
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Could not validate credentials",
        )

    user = await db.get(User, token_data.sub)
    if not user:
        raise HTTPException(status_code=404, detail="User not found")
    if not user.is_active:
        raise HTTPException(status_code=400, detail="Inactive user")
    return user

4. ## Common Anti-Patterns & Gotchas to Avoid

1. Mixing Synchronous Operations with Async Loops:

Anti-Pattern:* Calling blocking I/O libraries (e.g., standard requests, synchronous database drivers like psycopg2) inside async path_operation functions. This blocks the main event loop. Correct Approach:* Always use asynchronous drivers (asyncpg, httpx) or offload blocking tasks via run_in_threadpool.

2. Global Database Session Management:

Anti-Pattern:* Instantiating a single global SQLAlchemy Session instance and importing it across multiple modules. This leads to thread-safety violations and unhandled transaction contamination. Correct Approach:* Use FastAPI's dependency injection system with yield to scope sessions cleanly to the lifetime of each individual HTTP request.

3. Exposing Raw Database Models in API Responses:

Anti-Pattern:* Returning raw SQLAlchemy ORM entities directly from route handlers. This leaks sensitive attributes (like password hashes) and creates tight coupling between internal DB schemas and public API contracts. Correct Approach:* Always serialize models through Pydantic response schemas (response_model=UserResponse).

4. Hardcoding Secrets & Configuration:

Anti-Pattern:* Embedding JWT secrets, database connection strings, or third-party API keys directly into source code files. Correct Approach:* Enforce environment-based configuration strictly via Pydantic-Settings (BaseSettings).

5. ## Performance & Optimization Tips

  • Uvicorn Worker Scaling: In production container environments, run Uvicorn behind a process manager or configure multiple worker processes (uvicorn app.main:app --workers 4) to fully saturate multi-core CPU nodes.
  • Connection Pooling tuning: Tune SQLAlchemy engine parameters (pool_size, max_overflow, pool_timeout) relative to your database's maximum allowed connections and expected peak concurrent request load.
  • Query Optimization: Prevent N+1 query performance bottlenecks by utilizing SQLAlchemy's selectinload or joinedload strategies for relationship mapping.
  • Response Compression: Integrate Gzip middleware (from fastapi.middleware.gzip import GzipMiddleware) to minimize payload sizes for large JSON list endpoints.
