    POSTGRES_DB: str

    model_config = SettingsConfigDict(


        env_file=".env", 


        env_file_encoding="utf-8", 


        extra="ignore"


    )

        if not self.DATABASE_URL:

                f"postgresql+asyncpg://{self.POSTGRES_USER}:"


                f"{self.POSTGRES_PASSWORD}@{self.POSTGRES_SERVER}/{self.POSTGRES_DB}"


            )

2. Asynchronous Database Engine & Session (app/core/database.py)

from typing import AsyncGenerator

from sqlalchemy.ext.asyncio import AsyncSession, async_sessionmaker, create_async_engine


from app.core.config import settings



engine = create_async_engine(


    settings.DATABASE_URL,


    echo=False,


    future=True,


    pool_size=20,


    max_overflow=10,


)



AsyncSessionLocal = async_sessionmaker(


    bind=engine,


    class_=AsyncSession,


    expire_on_commit=False,


    autoflush=False,


)



async def get_db() -> AsyncGenerator[AsyncSession, None]:


    async with AsyncSessionLocal() as session:


        try:


            yield session


            await session.commit()


        except Exception:


            await session.rollback()


            raise


        finally:


            await session.close()

3. Cryptographic & JWT Security Utilities (app/core/security.py)

from datetime import datetime, timedelta, timezone

from typing import Any, Union


from jose import jwt


from passlib.context import CryptContext


from app.core.config import settings



pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")



def verify_password(plain_password: str, hashed_password: str) -> bool:


    return pwd_context.verify(plain_password, hashed_password)



def get_password_hash(password: str) -> str:


    return pwd_context.hash(password)



def create_access_token(subject: Union[str, Any], expires_delta: timedelta = None) -> str:


    if expires_delta:


        expire = datetime.now(timezone.utc) + expires_delta


    else:


        expire = datetime.now(timezone.utc) + timedelta(minutes=settings.ACCESS_TOKEN_EXPIRE_MINUTES)

    

    to_encode = {"exp": expire, "sub": str(subject), "type": "access"}


    encoded_jwt = jwt.encode(to_encode, settings.SECRET_KEY, algorithm=settings.ALGORITHM)


    return encoded_jwt



def create_refresh_token(subject: Union[str, Any]) -> str:


    expire = datetime.now(timezone.utc) + timedelta(days=settings.REFRESH_TOKEN_EXPIRE_DAYS)


    to_encode = {"exp": expire, "sub": str(subject), "type": "refresh"}


    encoded_jwt = jwt.encode(to_encode, settings.SECRET_KEY, algorithm=settings.ALGORITHM)


    return encoded_jwt

4. FastAPI Dependency Injection Pipeline (app/api/deps.py)

from fastapi import Depends, HTTPException, status

from fastapi.security import OAuth2PasswordBearer


from jose import JWTError, jwt


from sqlalchemy.ext.asyncio import AsyncSession


from sqlalchemy.future import select



from app.core.config import settings


from app.core.database import get_db


from app.models.user import User


from app.schemas.auth import TokenPayload



reusable_oauth2 = OAuth2PasswordBearer(


    tokenUrl=f"{settings.API_V1_STR}/auth/login"


)



async def get_current_user(


    db: AsyncSession = Depends(get_db),


    token: str = Depends(reusable_oauth2)


) -> User:


    credentials_exception = HTTPException(


        status_code=status.HTTP_401_UNAUTHORIZED,


        detail="Could not validate credentials",


        headers={"WWW-Authenticate": "Bearer"},


    )


    try:


        payload = jwt.decode(token, settings.SECRET_KEY, algorithms=[settings.ALGORITHM])


        user_id: str = payload.get("sub")


        token_type: str = payload.get("type")

        

        if user_id is None or token_type != "access":


            raise credentials_exception


        token_data = TokenPayload(sub=user_id)


    except JWTError:


        raise credentials_exception



    result = await db.execute(select(User).where(User.id == int(token_data.sub)))


    user = result.scalars().first()

    

    if user is None:


        raise credentials_exception


    if not user.is_active:


        raise HTTPException(status_code=400, detail="Inactive user account")

        

    return user

4. Common Anti-Patterns & Gotchas to Avoid

1. ❌ Blocking I/O in Async Routes:

The Mistake:* Using blocking libraries like standard requests, time.sleep(), or synchronous database calls (psycopg2) inside async def route handlers. This locks up the entire Uvicorn event loop worker. The Fix:* Use asynchronous drivers (asyncpg), httpx for outbound network requests, and asyncio.sleep(). If legacy sync libraries must be used, declare routes with regular def so FastAPI automatically offloads them to a threadpool.

2. ❌ Hardcoding Secret Keys:

The Mistake:* Embedding JWT SECRET_KEY variables directly in code or committing .env files to public version control. The Fix:* Always ingest configuration through Pydantic BaseSettings referencing environment variables injected securely via Docker secrets, AWS Secrets Manager, or HashiCorp Vault.

3. ❌ Ignoring Connection Pool Exhaustion:

The Mistake:* Instantiating a new database engine or session handler on every incoming request without proper closure. The Fix:* Utilize a singleton engine pattern combined with context-manager dependency injection (yield session) to guarantee automatic connection release.

4. ❌ Database Migrations on App Startup:

The Mistake:* Running alembic upgrade head programmatically inside main.py upon container boot in horizontally scaled multi-container clusters. The Fix:* Execute database migrations as a separate initialization job or pre-deployment task in your CI/CD or orchestrator pipeline (e.g., Kubernetes InitContainers or ECS task definitions).

5. Performance & Optimization Tips

Uvicorn Worker Scaling: Always configure Uvicorn with multiple workers based on CPU cores (--workers 4 or WEB_CONCURRENCY=2 CPU_COUNT + 1). When deploying behind a reverse proxy like Nginx or Traefik, leverage standard Unix domain sockets or internal container routing.
  • Database Indexing Strategy: Ensure foreign keys, lookup columns (email, username), and high-frequency filtering fields have appropriate B-Tree or Hash indexes defined in your SQLAlchemy models.
  • Response Compression: Integrate Starlette's GZipMiddleware to compress large JSON responses payloads exceeding 1000 bytes, drastically reducing bandwidth overhead over public networks.
  • Connection Keep-Alive: Configure upstream load balancers and reverse proxies with appropriate Keep-Alive timeouts to minimize TCP handshake overhead on high-frequency API calls.
