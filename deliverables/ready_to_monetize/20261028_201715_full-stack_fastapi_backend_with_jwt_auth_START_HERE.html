        finally:

from datetime import datetime, timedelta
from typing import Any, Union
from jose import jwt
from passlib.context import CryptContext
from app.core.config import settings

    return pwd_context.verify(plain_password, hashed_password)

    return pwd_context.hash(password)

def create_access_token(subject: Union[str, Any], expires_delta: timedelta = None) -> str:
    if expires_delta:
        expire = datetime.utcnow() + expires_delta
    else:
        expire = datetime.utcnow() + timedelta(minutes=settings.ACCESS_TOKEN_EXPIRE_MINUTES)
    
    to_encode = {"exp": expire, "sub": str(subject)}
    encoded_jwt = jwt.encode(to_encode, settings.SECRET_KEY, algorithm=settings.ALGORITHM)
    return encoded_jwt

from datetime import timedelta
from fastapi import APIRouter, Depends, HTTPException, status
from fastapi.security import OAuth2PasswordRequestForm
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy.future import select

from app.core.database import get_db
from app.core.security import verify_password, create_access_token
from app.core.config import settings
from app.models.user import User
from app.schemas.token import Token

router = APIRouter()

@router.post("/login/access-token", response_model=Token)
async def login_access_token(
    db: AsyncSession = Depends(get_db),
    form_data: OAuth2PasswordRequestForm = Depends()
) -> Any:
    """OAuth2 compatible token login, get an access token for future requests."""
    result = await db.execute(select(User).where(User.email == form_data.username))
    user = result.scalars().first()
    
    if not user or not verify_password(form_data.password, user.hashed_password):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Incorrect email or password",
            headers={"WWW-Authenticate": "Bearer"},
        )
    elif not user.is_active:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST, 
            detail="Inactive user account"
        )
        
    access_token_expires = timedelta(minutes=settings.ACCESS_TOKEN_EXPIRE_MINUTES)
    return {
        "access_token": create_access_token(user.id, expires_delta=access_token_expires),
        "token_type": "bearer",
    }

4. Common Anti-Patterns & Gotchas to Avoid

1. Blocking I/O in Async Route Handlers (def vs async def)

The Antipattern:* Using standard synchronous database drivers, standard file I/O operations, or blocking network requests inside an async def path operation. This blocks the main event loop thread and throttles all concurrent requests. The Fix: Use async def only* when executing asynchronous calls (like SQLAlchemy async sessions or httpx). If legacy libraries must be used, define the path operation with standard def so FastAPI executes it in an external thread pool.

2. Hardcoding Database Session Management

The Antipattern:* Manually initializing sessions within endpoints (session = SessionLocal()) without exception-safe try/except/finally teardowns or relying on auto-commits. The Fix:* Always use FastAPI's dependency injection container (Depends(get_db)) to ensure guaranteed transaction boundaries and leak-free connection pool handling.

3. Exposing Internal ORM Models directly to API Clients

The Antipattern:* Returning raw SQLAlchemy database models directly from paths, exposing internal database schemas, password hashes, and relationship cascades. The Fix:* Always map database models to dedicated Pydantic output schemas (response_model=UserPublicSchema) to enforce payload sanitization.

5. Performance & Optimization Tips

  • Uvicorn Worker Scaling: In production, do not rely on a single process instance. Run Uvicorn with multiple worker processes behind an Nginx or Traefik reverse proxy based on your available CPU vCores:
uvicorn app.main:app --workers 4 --loop uvloop --http llhttp
  • Connection Pooling: Fine-tune your SQLAlchemy pool configurations (pool_size, max_overflow, pool_timeout) to match your PostgreSQL instance's max_connections ceiling.
  • Database Indexing: Ensure all foreign keys, unique email fields, and filter parameters frequently hit in WHERE clauses carry explicit indexes within Alembic migration scripts.
  • Response Compression: Implement GZipMiddleware in app/main.py for payload outputs larger than 1000 bytes to preserve egress bandwidth:
from fastapi.middleware.gzip import GZipMiddleware

  app.add_middleware(GZipMiddleware, minimum_size=1000)
