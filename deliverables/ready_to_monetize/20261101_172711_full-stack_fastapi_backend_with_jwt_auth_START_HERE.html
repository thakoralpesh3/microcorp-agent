from collections.abc import AsyncGenerator
from sqlalchemy.ext.asyncio import AsyncSession, async_sessionmaker, create_async_engine
from app.core.config import settings

engine = create_async_engine(
    settings.DATABASE_URL, 
    echo=False, 
    future=True,
    pool_pre_ping=True
)

SessionLocal = async_sessionmaker(
    bind=engine, 
    class_=AsyncSession, 
    expire_on_commit=False,
    autocommit=False,
    autoflush=False
)

async def get_db() -> AsyncGenerator[AsyncSession, None]:
    async with SessionLocal() as session:
        try:
            yield session
            await session.commit()
        except Exception:
            await session.rollback()
            raise
        finally:
            await session.close()

from datetime import datetime, timedelta
from jose import jwt
from passlib.context import CryptContext
from app.core.config import settings

pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")

def verify_password(plain_password: str, hashed_password: str) -> bool:
    return pwd_context.verify(plain_password, hashed_password)

def get_password_hash(password: str) -> str:
    return pwd_context.hash(password)

def create_access_token(subject: str | int, expires_delta: timedelta | None = None) -> str:
    if expires_delta:
        expire = datetime.utcnow() + expires_delta
    else:
        expire = datetime.utcnow() + timedelta(minutes=settings.ACCESS_TOKEN_EXPIRE_MINUTES)
    
    to_encode = {"exp": expire, "sub": str(subject)}
    encoded_jwt = jwt.encode(to_encode, settings.SECRET_KEY, algorithm=settings.ALGORITHM)
    return encoded_jwt

from fastapi import Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer
from jose import JWTError, jwt
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy.future import select

from app.core.config import settings
from app.core.database import get_db
from app.models.user import User
from app.schemas.auth import TokenPayload

reusable_oauth2 = OAuth2PasswordBearer(
    tokenUrl=f"{settings.API_V1_STR}/auth/login"
)

async def get_current_user(
    db: AsyncSession = Depends(get_db),
    token: str = Depends(reusable_oauth2)
) -> User:
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Could not validate credentials",
        headers={"WWW-Authenticate": "Bearer"},
    )
    try:
        payload = jwt.decode(token, settings.SECRET_KEY, algorithms=[settings.ALGORITHM])
        user_id: str = payload.get("sub")
        if user_id is None:
            raise credentials_exception
        token_data = TokenPayload(sub=user_id)
    except JWTError:
        raise credentials_exception
        
    result = await db.execute(select(User).where(User.id == int(token_data.sub)))
    user = result.scalars().first()
    
    if not user:
        raise credentials_exception
    if not user.is_active:
        raise HTTPException(status_code=400, detail="Inactive user")
        
    return user

async def get_current_active_superuser(
    current_user: User = Depends(get_current_user),
) -> User:
    if not current_user.is_superuser:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="The user doesn't have enough privileges"
        )
    return current_user

from fastapi import APIRouter, Depends, HTTPException, status
from fastapi.security import OAuth2PasswordRequestForm
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy.future import select

from app.api.deps import get_db
from app.core.security import create_access_token, verify_password
from app.models.user import User
from app.schemas.auth import Token

router = APIRouter()

@router.post("/login", response_model=Token)
async def login_access_token(
    db: AsyncSession = Depends(get_db), 
    form_data: OAuth2PasswordRequestForm = Depends()
):
    result = await db.execute(select(User).where(User.email == form_data.username))
    user = result.scalars().first()
    
    if not user or not verify_password(form_data.password, user.hashed_password):
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST, 
            detail="Incorrect email or password"
        )
    elif not user.is_active:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST, 
            detail="Inactive user"
        )
        
    access_token = create_access_token(subject=user.id)
    return {"access_token": access_token, "token_type": "bearer"}

4. Common Anti-Patterns & Gotchas to Avoid

1. Mixing Synchronous Operations in Async Endpoints

Anti-Pattern:* Calling blocking I/O libraries (like standard requests or blocking DB calls like standard psycopg2) inside an async def path operation function. This freezes the entire event loop thread. Solution:* Use async-native drivers (asyncpg, httpx) or offload blocking CPU-heavy operations via run_in_threadpool.

2. Hardcoding Secrets or Settings Configuration

Anti-Pattern:* Keeping JWT secret keys or database passwords hardcoded in python source code files. Solution:* Always use Pydantic BaseSettings reading environment variables through a securely guarded .env file excluded from version control via .gitignore.

3. Not Using Connection Pool Ping (pool_pre_ping=True)

Anti-Pattern:* Creating database engines without connection health checks, leading to dropped connection exceptions after periods of database inactivity (OperationalError: server closed the connection unexpectedly). Solution:* Always instantiate SQLAlchemy engines with pool_pre_ping=True to automatically validate connections before query execution.

4. Exposing ORM Models Directly as Responses

Anti-Pattern:* Returning raw SQLAlchemy database models directly from FastAPI endpoints. This can inadvertently expose sensitive internal attributes (such as hashed_password) or trigger lazy-loading recursion errors. Solution:* Always map your database models to explicitly declared Pydantic response schemas (response_model=UserResponse).

5. Performance & Optimization Tips

  • Uvicorn Worker Scaling via Gunicorn: In production environments, run Uvicorn with multiple worker processes using Gunicorn as a process manager to utilize all CPU cores:
gunicorn app.main:app -w 4 -k uvicorn.workers.UvicornWorker --bind 0.0.0.0:8000
  • Response Compression: Add Starlette's GZipMiddleware to compress large JSON responses automatically and reduce network payload overhead:
from fastapi.middleware.gzip import GZipMiddleware

  app.add_middleware(GZipMiddleware, minimum_size=1000)

  • Database Indexing: Ensure frequently queried relational fields (e.g., email, foreign key IDs) have explicit database indices defined in your SQLAlchemy models (index=True).
  • Dependency Caching (lru_cache): Cache configuration settings instantiation using @lru_cache() so that environment checks do not repeatedly re-read the .env disk storage on every request cycle.
